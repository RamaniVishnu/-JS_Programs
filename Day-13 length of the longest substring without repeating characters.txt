let str = "abbbcabcdefef";

function longestValue(data) {
    if (!data) return "";

    let output = "";
    let count = 1;

    for (let i = 1; i <= data.length; i++) {
        if (data[i] === data[i - 1]) {
            count++;
        } else {
            output += data[i - 1] + count;
            count = 1;
        }
    }

    return output;
}

console.log(longestValue(str));
