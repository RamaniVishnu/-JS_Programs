//enhanced code, pure function, reusable   ->concat() creates new array everytime so going with 2 functions is best practice 
function flatArray(data) { 
    let result = []; 
    data.forEach((elem) => { 
        if (Array.isArray(elem)) { 
            result = result.concat(flatArray(elem));
        }
        else {
        result.push(elem);
        }
        });
        return result; 
    
}
